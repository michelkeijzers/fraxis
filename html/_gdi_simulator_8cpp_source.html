132{
133 switch (message)

138
139 case WM_LCD_1602_DISPLAY_UPDATE:

142
143 case WM_TM1637_UPDATE:
144 _gdiScreen->UpdateTm1637();
145 break;
146
147 case WM_MOUSEMOVE:
148 {
149 int mx = GET_X_LPARAM(lParam);
150 int my = GET_Y_LPARAM(lParam);
151
152 _gdiScreen->OnMouseMove(mx, my);
153 InvalidateRect(hWnd, NULL, FALSE);
154 }
155 break;
156
157 case WM_LBUTTONDOWN:
158 {
159 int mx = GET_X_LPARAM(lParam);
160 int my = GET_Y_LPARAM(lParam);
161
162 _gdiScreen->OnMouseDown(mx, my);
163 InvalidateRect(hWnd, NULL, FALSE);
164 }
165 break;
166
167 case WM_LBUTTONUP:
168 {
169 int mx = GET_X_LPARAM(lParam);
170 int my = GET_Y_LPARAM(lParam);
171
172 _gdiScreen->OnMouseUp(mx, my);
173 InvalidateRect(hWnd, NULL, FALSE);
174 }
175 break;
176
177 case WM_KEYDOWN:
178 {
179 WindowsMcp23017* mcp23017 = dynamic_cast<WindowsMcp23017*>(_drivers.mcp23017);
180 switch (wParam)
181 {
182 case VK_ESCAPE: mcp23017->SimulateSetGpioPin(PinIoMappings::EIdBit::SystemButton , 1); break;
183 case VK_SPACE: mcp23017->SimulateSetGpioPin(PinIoMappings::EIdBit::Player1Button , 1); break;
184 case VK_UP: mcp23017->SimulateSetGpioPin(PinIoMappings::EIdBit::Player1Up , 1); break;
185 case VK_DOWN: mcp23017->SimulateSetGpioPin(PinIoMappings::EIdBit::Player1Down , 1); break;
186 case VK_LEFT: mcp23017->SimulateSetGpioPin(PinIoMappings::EIdBit::Player1Left , 1); break;
187 case VK_RIGHT: mcp23017->SimulateSetGpioPin(PinIoMappings::EIdBit::Player1Right , 1); break;
188
189 case VK_NUMPAD5: mcp23017->SimulateSetGpioPin(PinIoMappings::EIdBit::Player2Button , 1); break;
190 case VK_NUMPAD8: mcp23017->SimulateSetGpioPin(PinIoMappings::EIdBit::Player2Up , 1); break;
191 case VK_NUMPAD2: mcp23017->SimulateSetGpioPin(PinIoMappings::EIdBit::Player2Down , 1); break;
192 case VK_NUMPAD4: mcp23017->SimulateSetGpioPin(PinIoMappings::EIdBit::Player2Left , 1); break;
193 case VK_NUMPAD6: mcp23017->SimulateSetGpioPin(PinIoMappings::EIdBit::Player2Right , 1); break;
194 }
195 }
196 break;
197
198 case WM_KEYUP:
199 {
200 WindowsMcp23017* mcp23017 = dynamic_cast<WindowsMcp23017*>(_drivers.mcp23017);
201 mcp23017->SimulateResetGpioPins();
202 }
203 break;
204
205 case WM_PAINT:
206 {
207 PAINTSTRUCT ps;
208 HDC hdc = BeginPaint(hWnd, &ps);
209
210 RECT rc;

212 BitBlt(hdc, 0, 0, rc.right - rc.left, rc.bottom - rc.top, _gdiScreen->GetMemDC(), 0, 0, SRCCOPY);
213 EndPaint(hWnd, &ps);
214 }
215 break;
216
217 case WM_ERASEBKGND:
218 return 1; // Prevent flickering by not erasing the background
219
220 case WM_TIMER:
221 _gdiScreen->Update();
222 InvalidateRect(hWnd, NULL, FALSE); // request redraw
223 break;
224
225 case WM_DESTROY:

void DelayUs(uint32_t us)
{
    auto start = std::chrono::high_resolution_clock::now();
    while (std::chrono::duration_cast<std::chrono::microseconds>(
               std::chrono::high_resolution_clock::now() - start).count() < us)
    {
        // busy-wait
    }
}
